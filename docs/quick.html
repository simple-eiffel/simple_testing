Quick Examples

Basic Test Class

class MY_TEST_SET
inherit TEST_SET_BASE
feature
    test_basic_assertion
        do
            assert ("condition_true", True)
            assert ("condition_false", not False)
        end
end

System-Level Fixture (Singleton)

class MY_APP_TEST_SYSTEM
inherit TEST_SYSTEM
feature
    on_prepare_test_system
        do
            -- Initialize database, server, pools
            -- Called ONCE at test run start
        end

    on_clean_test_system
        do
            -- Cleanup resources
            -- Called ONCE at test run end
        end
end

Feature-Level Fixture (Per-Test)

test_with_fixture
    local
        l_fixture: TEST_FIXTURE_MOCK
    do
        create l_fixture
        l_fixture.on_create
        -- Test logic here
        assert ("is_valid", l_fixture.is_valid)
        l_fixture.on_destroy
    end

Nested Fixtures with Scoped Cleanup

test_nested_fixture
    local
        l_test: FIXTURE_HOLDER
    do
        create l_test
        l_test.with_fixture (agent do
            -- Fixture acquired here
            -- Cleanup automatic on exit
            assert ("in_scope", True)
        end)
    end

Common Assertions

-- Boolean assertions
assert ("is_true", some_condition)
assert_true ("explicit_true", condition)
assert_false ("explicit_false", not condition)

-- Object assertions
assert_attached ("not_void", result)
assert_void ("is_void", optional_value)
assert_same_reference ("same_obj", obj1, obj2)

-- Equality assertions
assert_equal ("equals", expected, actual)
assert_not_equal ("not_equals", x, y)

-- Numeric assertions
assert_positive ("positive", count)
assert_negative ("negative", delta)
assert_in_range ("in_range", value, 1, 100)

-- String assertions
assert_string_contains ("has_text", output, "ERROR")
assert_string_starts_with ("header", response, "HTTP")
assert_string_empty ("empty_str", empty_var)

-- Collection assertions
assert_count_equals ("size", 5, my_list)
assert_iterable_not_empty ("has_items", collection)

Common Operations

State Transition Verification

test_state_transition
    local
        l_system: TEST_SYSTEM_MOCK
        old_state, new_state: INTEGER
    do
        create l_system
        old_state := l_system.initialization_state
        l_system.on_prepare_test_system
        new_state := l_system.initialization_state
        assert_equal ("transition_0_to_1", new_state, 1)
    end

Model Query Verification (MML)

test_model_query
    local
        l_system: TEST_SYSTEM_MOCK
    do
        create l_system
        assert_equal ("uninitialized_state", 0, l_system.initialization_state)
        l_system.on_prepare_test_system
        assert_equal ("initialized_state", 1, l_system.initialization_state)
    end

Exception-Safe Cleanup

test_exception_safety
    local
        l_fixture: TEST_FIXTURE_MOCK
    do
        create l_fixture
        l_fixture.on_create
        -- Even if test logic throws, on_destroy guarantees cleanup
        -- Eiffel's rescue clause ensures postcondition holds
        l_fixture.on_destroy
        assert ("cleaned_up", not l_fixture.is_valid)
    end
